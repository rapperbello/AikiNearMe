import {getProviders,saveProvider,read,write,escapeHTML,money} from "./data.js"; const $=id=>document.getElementById(id); function render(){ const ps=getProviders(),bs=read("bookings",[]),rs=read("reports",[]),ss=read("subscriptions",[]); $("adminStats").innerHTML=[["Providers",ps.length],["Bookings",bs.length],["Reports",rs.filter(r=>r.status==="Pending").length],["Recorded revenue",money(ss.filter(s=>s.status==="Verified paid").reduce((n,s)=>n+s.amount,0))]].map(([label,value])=>`
${label}${value}
`).join(""); $("adminProviders").innerHTML=ps.length?ps.map(p=>`
${escapeHTML(p.name)}

${escapeHTML(p.service)} · ${escapeHTML(p.location)} · ${money(p.price)}

${p.verified?"Verified":"Not verified"} · ${p.premium?"Premium":"Free"}

`).join(""):'

No providers.

'; $("adminBookings").innerHTML=bs.length?bs.map(b=>`
${escapeHTML(b.service)}

${escapeHTML(b.providerName)} · ${escapeHTML(b.date)} · ${escapeHTML(b.location)}

${escapeHTML(b.status)}
`).join(""):'

No bookings.

'; $("adminReports").innerHTML=rs.length?rs.map(r=>`
${escapeHTML(r.providerName||"Provider report")}

${escapeHTML(r.reason)}

${escapeHTML(r.status)}
`).join(""):'

No reports.

'; $("adminProviders").querySelectorAll("[data-verify]").forEach(b=>b.onclick=()=>{const p=getProviders().find(x=>x.id===b.dataset.verify);if(p)saveProvider({...p,verified:!p.verified});render()}); $("adminProviders").querySelectorAll("[data-premium]").forEach(b=>b.onclick=()=>{const p=getProviders().find(x=>x.id===b.dataset.premium);if(p)saveProvider({...p,premium:!p.premium});render()}); $("adminProviders").querySelectorAll("[data-remove]").forEach(b=>b.onclick=()=>{if(confirm("Remove this provider from this browser demo?")){write("providers",getProviders().filter(p=>p.id!==b.dataset.remove));render()}}); $("adminReports").querySelectorAll("[data-report]").forEach(b=>b.onclick=()=>{write("reports",rs.map(r=>r.id===b.dataset.report?{...r,status:"Reviewed"}:r));render()}); } $("exportData").onclick=()=>{const data={providers:getProviders(),bookings:read("bookings",[]),reports:read("reports",[]),subscriptions:read("subscriptions",[]),reviews:read("reviews",[])};const blob=new Blob([JSON.stringify(data,null,2)],{type:"application/json"});const a=document.createElement("a");a.href=URL.createObjectURL(blob);a.download="aikinearme-demo-data.json";a.click();URL.revokeObjectURL(a.href)}; $("clearDemoData").onclick=()=>{if(confirm("Clear demo data from this browser? This cannot be undone.")){["ainm_providers","ainm_bookings","ainm_reports","ainm_blocked","ainm_subscriptions","ainm_notifications","ainm_reviews"].forEach(k=>localStorage.removeItem(k));localStorage.removeItem("ainm_my_provider_id");location.reload()}}; render();
